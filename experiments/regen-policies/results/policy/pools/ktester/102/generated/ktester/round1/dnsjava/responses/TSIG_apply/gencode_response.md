package org.xbill.DNS;

import static org.junit.jupiter.api.Assertions.*;

import java.lang.reflect.Field;
import java.time.Clock;
import java.time.Instant;
import java.time.ZoneId;
import java.util.List;
import javax.crypto.SecretKey;
import javax.crypto.spec.SecretKeySpec;

import org.junit.jupiter.api.*;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;

/**
 * Unit‑tests for {@link TSIG#apply(Message, TSIGRecord)}.
 *
 * <p>The tests cover the exhaustive set of scenarios described in the JSON test‑case
 * specification.  Each test focuses on the observable side‑effects of the method
 * (presence of a TSIG record, message‑signed flag, algorithm/key name, and expected
 * exceptions).  A deterministic {@link Clock} is used so that the {@code timeSigned}
 * field of generated {@link TSIGRecord}s is predictable.
 */
@ExtendWith(MockitoExtension.class)
class TSIG_apply_Test {

    /** TSIG instance used for all tests – created with a deterministic clock. */
    private static TSIG tsig;

    /** Fixed clock to make the signature timestamp predictable. */
    private static final Clock FIXED_CLOCK =
            Clock.fixed(Instant.parse("2023-01-01T00:00:00Z"), ZoneId.of("UTC"));

    /** Simple base‑64 encoded key (the string “1234567890”). */
    private static final String B64_KEY = "MTIzNDU2Nzg5MA==";

    @BeforeAll
    static void setupBeforeAll() {
        // algorithm = HMAC‑SHA256, name = "example."
        Name algorithm = TSIG.HMAC_SHA256;
        Name name = Name.fromString("example.");
        SecretKey key = new SecretKeySpec(base64.fromString(B64_KEY), "HmacSHA256");
        // The TSIG constructor used by the original test suite (algorithm, name, key, clock)
        tsig = new TSIG(algorithm, name, key, FIXED_CLOCK);
    }

    /* --------------------------------------------------------------------- */
    /* Helper utilities                                                    */
    /* --------------------------------------------------------------------- */

    /**
     * Renders a {@link Message} to wire format.  The {@code Message} class in dnsjava
     * provides {@code toWire()} which returns a byte[].  If the method does not exist
     * we fall back to {@code toWire(Message.maxLength)} – the exact signature is not
     * important for the tests because the generated bytes are only used as input to
     * {@link TSIG#generate}.
     */
    private static byte[] render(Message m) {
        try {
            return (byte[]) Message.class.getMethod("toWire").invoke(m);
        } catch (ReflectiveOperationException e) {
            // fallback – most dnsjava versions have toWire(int)
            try {
                return (byte[]) Message.class.getMethod("toWire", int.class)
                        .invoke(m, 512);
            } catch (ReflectiveOperationException ex) {
                throw new AssertionError("Unable to render Message to wire format", ex);
            }
        }
    }

    /**
     * Creates a {@link TSIGRecord} with the supplied parameters using reflection.
     * The constructor signature of {@code TSIGRecord} in dnsjava is:
     *
     * <pre>
     * TSIGRecord(Name name, int dclass, long ttl, Name algorithm,
     *            Instant timeSigned, int fudge, byte[] mac, byte[] other)
     * </pre>
     *
     * If the constructor changes, the reflective code will fail, causing the test
     * suite to surface the incompatibility.
     */
    private static TSIGRecord createTsigRecord(Name name,
                                                Name algorithm,
                                                Instant timeSigned,
                                                byte[] mac,
                                                byte[] other) {
        try {
            Class<?>[] paramTypes = new Class<?>[]{
                    Name.class, int.class, long.class, Name.class,
                    Instant.class, int.class, byte[].class, byte[].class
            };
            // dclass = DClass.IN, ttl = 0, fudge = (int)TSIG.FUDGE.getSeconds()
            Object record = TSIGRecord.class
                    .getDeclaredConstructor(paramTypes)
                    .newInstance(name, DClass.IN, 0L, algorithm,
                            timeSigned, (int) TSIG.FUDGE.getSeconds(),
                            mac, other);
            return (TSIGRecord) record;
        } catch (ReflectiveOperationException e) {
            throw new AssertionError("Failed to create TSIGRecord via reflection", e);
        }
    }

    /**
     * Alters the {@code algorithm} field of an existing {@link TSIGRecord} using reflection.
     */
    private static void setAlgorithm(TSIGRecord rec, Name newAlg) {
        try {
            Field algField = TSIGRecord.class.getDeclaredField("algorithm");
            algField.setAccessible(true);
            algField.set(rec, newAlg);
        } catch (ReflectiveOperationException e) {
            throw new AssertionError("Unable to set algorithm via reflection", e);
        }
    }

    /**
     * Alters the {@code key} (the secret) of a {@link TSIGRecord}.  The key bytes are
     * stored in the private {@code mac} field for the purpose of these tests – the
     * actual implementation uses the key from the {@link TSIG} instance, but we need a
     * record that pretends to have a different key for a few edge cases.
     */
    private static void setMac(TSIGRecord rec, byte[] mac) {
        try {
            Field macField = TSIGRecord.class.getDeclaredField("mac");
            macField.setAccessible(true);
            macField.set(rec, mac);
        } catch (ReflectiveOperationException e) {
            throw new AssertionError("Unable to set MAC via reflection", e);
        }
    }

    /* --------------------------------------------------------------------- */
    /* Test cases – one method per JSON group                              */
    /* --------------------------------------------------------------------- */

    /**
     * Group: apply(Message, TSIGRecord) - Null Message
     */
    @Test
    void applyRejectsNullMessage() {
        TSIGRecord dummyOld = tsig.generate(Message.newQuery(Record.newRecord(Name.fromString("a."), Type.A, DClass.IN)),
                new byte[0], Rcode.NOERROR, null, true);
        assertThrows(NullPointerException.class,
                () -> tsig.apply(null, dummyOld),
                "A null Message should cause NullPointerException");
    }

    /**
     * Group: apply(Message, TSIGRecord) - Null Old Record
     */
    @Test
    void applyCreatesTsigWhenOldIsNull() throws Exception {
        Record q = Record.newRecord(Name.fromString("www.example."), Type.A, DClass.IN);
        Message msg = Message.newQuery(q);
        assertFalse(msg.isSigned());

        tsig.apply(msg, null);

        TSIGRecord rec = msg.getTSIG();
        assertNotNull(rec, "TSIG record must be added");
        assertTrue(msg.isSigned(), "Message must be marked signed");
        assertEquals(TSIG.HMAC_SHA256, rec.getAlgorithm());
        assertEquals(Name.fromString("example."), rec.getName());
        assertEquals(Instant.parse("2023-01-01T00:00:00Z"), rec.getTimeSigned(),
                "Timestamp must come from fixed clock");
    }

    /**
     * Group: apply(Message, TSIGRecord) - Valid Old Record (full signature)
     */
    @Test
    void applyWithValidOldRecordFullSignature() throws Exception {
        // 1️⃣ Create a request and obtain its TSIG (old record)
        Record q = Record.newRecord(Name.fromString("www.example."), Type.A, DClass.IN);
        Message request = Message.newQuery(q);
        tsig.apply(request, null);
        TSIGRecord old = request.getTSIG();
        assertNotNull(old);

        // 2️⃣ Build a fresh response that references the request
        Message response = new Message();
        response.getHeader().setID(request.getHeader().getID());
        response.addRecord(q, Section.QUESTION);
        assertFalse(response.isSigned());

        // 3️⃣ Apply TSIG using the old record (fullSignature = true)
        tsig.apply(response, old);

        // 4️⃣ Verify
        TSIGRecord respTsig = response.getTSIG();
        assertNotNull(respTsig);
        assertTrue(response.isSigned());
        assertEquals(old.getAlgorithm(), respTsig.getAlgorithm());
        assertEquals(old.getName(), respTsig.getName());
        assertEquals(request.getHeader().getID(), respTsig.getOriginalID(),
                "Original ID must be preserved");
    }

    /**
     * Group: apply(Message, TSIGRecord) - Old Record with Different Algorithm
     */
    @Test
    void applyRejectsOldRecordWithUnsupportedAlgorithm() throws Exception {
        // Build a normal old record first
        Record q = Record.newRecord(Name.fromString("a.example."), Type.A, DClass.IN);
        Message req = Message.newQuery(q);
        tsig.apply(req, null);
        TSIGRecord old = req.getTSIG();

        // Change its algorithm to an unknown name
        Name unknownAlg = Name.fromString("unknown-alg.");
        setAlgorithm(old, unknownAlg);

        Message msg = Message.newQuery(q);
        assertThrows(IllegalArgumentException.class,
                () -> tsig.apply(msg, old),
                "Unsupported algorithm should cause IllegalArgumentException");
    }

    /**
     * Group: apply(Message, TSIGRecord) - Message Already Contains TSIG
     */
    @Test
    void applyReplacesExistingTsigWhenMessageAlreadyHasOne() throws Exception {
        Record q = Record.newRecord(Name.fromString("b.example."), Type.A, DClass.IN);
        Message msg = Message.newQuery(q);
        // First signing
        tsig.apply(msg, null);
        TSIGRecord first = msg.getTSIG();
        assertNotNull(first);

        // Second signing – old is null, should replace without error
        tsig.apply(msg, null);
        TSIGRecord second = msg.getTSIG();
        assertNotNull(second);
        assertNotSame(first, second, "A new TSIGRecord should replace the old one");
        assertEquals(1, msg.getSection(Section.ADDITIONAL).stream()
                .filter(r -> r.getType() == Type.TSIG).count(),
                "Exactly one TSIG record must remain");
    }

    /**
     * Group: apply(Message, TSIGRecord) - Edge Case: Empty Message Sections

    @Test
    void applyHandlesEmptyMessage() throws Exception {

        assertTrue(empty.isSigned());
    }

    /**
     * Group: apply(Message, TSIGRecord) - Existing TSIG with Non‑null Old Record (different key)
     */
    @Test
    void applyUsesOldRecordKeyWhenDifferentFromExisting() throws Exception {
        // Message with an existing TSIG (key "example.")
        Record q = Record.newRecord(Name.fromString("c.example."), Type.A, DClass.IN);
        Message msg = Message.newQuery(q);
        tsig.apply(msg, null);
        TSIGRecord existing = msg.getTSIG();
        assertNotNull(existing);

        // Create a second TSIG instance with a different key (keyB)
        SecretKey keyB = new SecretKeySpec(base64.fromString("YWJjZGVmZ2hpamtsbW5vcA=="), "HmacSHA256"); // "abcdefghijklmnop"
        TSIG tsigB = new TSIG(TSIG.HMAC_SHA256, Name.fromString("keyB."), keyB, FIXED_CLOCK);

        // Generate an old record using tsigB (different key)
        Message dummy = Message.newQuery(q);
        tsigB.apply(dummy, null);
        TSIGRecord oldFromB = dummy.getTSIG();

        // Apply using the old record from keyB
        tsig.apply(msg, oldFromB);
        TSIGRecord after = msg.getTSIG();
        assertNotNull(after);
        assertEquals(Name.fromString("keyB."), after.getName(),
                "The new TSIG should use the key name from the supplied old record");
    }

    /**
     * Group: apply(Message, TSIGRecord) - Old Record using HMAC‑MD5 algorithm
     */
    @Test
    void applySupportsLegacyHmacMd5Algorithm() throws Exception {
        // Build an old record that uses HMAC‑MD5
        Record q = Record.newRecord(Name.fromString("d.example."), Type.A, DClass.IN);
        Message req = Message.newQuery(q);
        // Use a TSIG instance configured with HMAC‑MD5 algorithm
        SecretKey md5Key = new SecretKeySpec(base64.fromString(B64_KEY), "HmacMD5");
        TSIG tsigMd5 = new TSIG(TSIG.HMAC_MD5, Name.fromString("example."), md5Key, FIXED_CLOCK);
        tsigMd5.apply(req, null);
        TSIGRecord oldMd5 = req.getTSIG();

        Message msg = Message.newQuery(q);
        tsig.apply(msg, oldMd5); // tsig (SHA256) should accept the old record's algorithm
        TSIGRecord newRec = msg.getTSIG();
        assertNotNull(newRec);
        assertEquals(TSIG.HMAC_MD5, oldMd5.getAlgorithm(),
                "Old record must retain HMAC‑MD5 algorithm");
        assertEquals(TSIG.HMAC_SHA256, newRec.getAlgorithm(),
                "Generated TSIG uses default algorithm (SHA256)");
    }

    /**
     * Group: apply(Message, TSIGRecord) - Old Record with Future Timestamp beyond fudge
     */
    @Test
    void applyRejectsOldRecordWithFutureTimestampBeyondFudge() throws Exception {
        Record q = Record.newRecord(Name.fromString("e.example."), Type.A, DClass.IN);
        Message req = Message.newQuery(q);
        tsig.apply(req, null);
        TSIGRecord old = req.getTSIG();

        // Push timestamp 10 minutes into the future
        Instant future = Instant.now().plusSeconds(600);
        setAlgorithm(old, TSIG.HMAC_SHA256); // ensure algorithm is valid
        // Use reflection to modify the private timeSigned field
        Field timeField = TSIGRecord.class.getDeclaredField("timeSigned");
        timeField.setAccessible(true);
        timeField.set(old, future);

        Message msg = Message.newQuery(q);
        assertThrows(SecurityException.class,
                () -> tsig.apply(msg, old),
                "Future timestamp beyond fudge should cause SecurityException");
    }

    /**
     * Group: apply(Message, TSIGRecord) - Old Record with Zero‑Length MAC
     */
    @Test
    void applyHandlesOldRecordWithZeroLengthMac() throws Exception {
        Record q = Record.newRecord(Name.fromString("f.example."), Type.A, DClass.IN);
        Message req = Message.newQuery(q);
        tsig.apply(req, null);
        TSIGRecord old = req.getTSIG();

        // Set MAC to zero‑length array
        setMac(old, new byte[0]);

        Message msg = Message.newQuery(q);
        // Should not throw – MAC length 0 is treated as empty
        assertDoesNotThrow(() -> tsig.apply(msg, old));
        TSIGRecord newRec = msg.getTSIG();
        assertNotNull(newRec);
    }

    /**

    @Test
    void applyWorksWithAllSectionsPopulated() throws Exception {
        // Build a message containing all four sections
        Name name = Name.fromString("full.example.");
        Record q = Record.newRecord(name, Type.A, DClass.IN);
        Record a = Record.newRecord(name, Type.A, DClass.IN, "1.2.3.4");

        Message msg = new Message();
        msg.addRecord(q, Section.QUESTION);
        msg.addRecord(a, Section.ANSWER);
        msg.addRecord(ns, Section.AUTHORITY);
        msg.addRecord(txt, Section.ADDITIONAL);

        tsig.apply(msg, null);
        TSIGRecord rec = msg.getTSIG();
        assertNotNull(rec);
        assertTrue(msg.isSigned());
    }

    @Test

        Record q = Record.newRecord(Name.fromString("g.example."), Type.A, DClass.IN);

    /**
     * Group: apply(Message, TSIGRecord) - Message containing duplicate TSIG records
     */
    @Test
    void applyDetectsDuplicateTsigRecords() throws Exception {
        // Build a message with two TSIG records manually
        Record q = Record.newRecord(Name.fromString("k.example."), Type.A, DClass.IN);
        Message msg = Message.newQuery(q);
        // First TSIG
        tsig.apply(msg, null);

        assertThrows(IllegalStateException.class,
                () -> tsig.apply(msg, null),
                "Duplicate TSIG records should cause IllegalStateException");
    }
